      function recompileLivePreview() {
        const html = generatePageHtml(activePageFile);
        lastGeneratedHtml = html;
        multiPageCache[activePageFile] = html;

        const iframe = document.getElementById('previewIframe');
        if (iframe) iframe.srcdoc = html;
        const code = document.getElementById('codeInspector');
        if (code && currentStageView === 'code') code.textContent = html;
      }

      function applyAiMutation() {
        const prompt = document.getElementById('aiMutationPrompt').value.trim();
        if (!prompt) return;

        if (window.ZothSiteSections && typeof window.ZothSiteSections.mutateSiteSection === 'function') {
          const mutated = window.ZothSiteSections.mutateSiteSection(lastGeneratedHtml, prompt, {
            name: document.getElementById('siteName').value.trim() || 'Apex Cloud',
            theme: currentTheme,
            niche: currentNiche
          });
          lastGeneratedHtml = mutated;
          const iframe = document.getElementById('previewIframe');
          if (iframe) iframe.srcdoc = mutated;
          const code = document.getElementById('codeInspector');
          if (code && currentStageView === 'code') code.textContent = mutated;
          alert('✨ AI Mutation Applied: ' + prompt);
        } else {
          recompileLivePreview();
          alert('✓ Recompiled with prompt: ' + prompt);
        }
      }

      function randomizeSite() {
        const niches = Object.keys(NICHES_DATA);
        const themes = Object.keys(THEMES_DATA);
        const randNiche = niches[Math.floor(Math.random() * niches.length)];
        const randTheme = themes[Math.floor(Math.random() * themes.length)];

        currentNiche = randNiche;
        currentTheme = randTheme;

        document.querySelectorAll('.niche-card').forEach(c => {
          if (c.onclick.toString().includes(randNiche)) c.classList.add('active');
          else c.classList.remove('active');
        });

        document.querySelectorAll('.theme-card').forEach(c => {
          if (c.onclick.toString().includes(randTheme)) c.classList.add('active');
          else c.classList.remove('active');
        });

        const adjectives = ['Quantum', 'Neural', 'Hyper', 'Cyber', 'Apex', 'Vortex', 'Sovereign', 'Aether'];
        const nouns = ['Cloud', 'Matrix', 'Pulse', 'Forge', 'Node', 'Nexus', 'Vault', 'Flow'];
        const randName = adjectives[Math.floor(Math.random() * adjectives.length)] + ' ' + nouns[Math.floor(Math.random() * nouns.length)];
        document.getElementById('siteName').value = randName;
        document.getElementById('siteTagline').value = '';

        recompileLivePreview();
      }

      }

      async function downloadFrameworkZip(type) {
        const name = document.getElementById('siteName').value.trim() || 'Apex Cloud';
        const slug = name.toLowerCase().replace(/[^a-z0-9]+/g, '-');

        if (type === 'single-html') {
          const blob = new Blob([lastGeneratedHtml], { type: 'text/html' });
          const a = document.createElement('a');
          a.href = URL.createObjectURL(blob);
          a.download = slug + '.html';
          a.click();
          return;
        }

        if (!window.JSZip) {
          alert('JSZip loading... please retry in 1 second.');
          return;
        }

        const zip = new JSZip();

        if (type === 'multipage-bundle') {
          ['index.html', 'features.html', 'pricing.html', 'docs.html', 'about.html', 'contact.html'].forEach(p => {
            zip.file(p, generatePageHtml(p));
          });
          zip.file('netlify.toml', '[[redirects]]\n  from = "/*"\n  to = "/index.html"\n  status = 200\n');
          const blob = await zip.generateAsync({ type: 'blob' });
          const a = document.createElement('a');
          a.href = URL.createObjectURL(blob);
          a.download = slug + '-multipage-bundle.zip';
          a.click();
        } else if (type === 'astro') {
          zip.file('package.json', JSON.stringify({ name: slug + '-astro', scripts: { dev: 'astro dev', build: 'astro build' }, dependencies: { astro: '^5.1.0', '@astrojs/tailwind': '^5.1.4', tailwindcss: '^3.4.17' } }, null, 2));
          zip.file('astro.config.mjs', "import { defineConfig } from 'astro/config';\nimport tailwind from '@astrojs/tailwind';\nexport default defineConfig({ integrations: [tailwind()] });\n");
          zip.file('src/pages/index.astro', "---\n---\n" + generatePageHtml('index.html'));
          const blob = await zip.generateAsync({ type: 'blob' });
          const a = document.createElement('a');
          a.href = URL.createObjectURL(blob);
          a.download = slug + '-astro-project.zip';
          a.click();
        } else if (type === 'vite-react') {
          zip.file('package.json', JSON.stringify({ name: slug + '-vite-react', scripts: { dev: 'vite', build: 'vite build' }, dependencies: { react: '^18.3.1', 'react-dom': '^18.3.1' }, devDependencies: { vite: '^6.0.0', '@vitejs/plugin-react': '^4.3.4', tailwindcss: '^3.4.17' } }, null, 2));
          zip.file('index.html', lastGeneratedHtml);
          const blob = await zip.generateAsync({ type: 'blob' });
          const a = document.createElement('a');
          a.href = URL.createObjectURL(blob);
          a.download = slug + '-vite-react-project.zip';
          a.click();
        } else if (type === 'nextjs') {
          zip.file('package.json', JSON.stringify({ name: slug + '-nextjs', scripts: { dev: 'next dev', build: 'next build', start: 'next start' }, dependencies: { next: '^15.1.0', react: '^19.0.0', 'react-dom': '^19.0.0' } }, null, 2));
          zip.file('app/page.tsx', "export default function Page() { return (
); }\n");
          const blob = await zip.generateAsync({ type: 'blob' });
          const a = document.createElement('a');
          a.href = URL.createObjectURL(blob);
          a.download = slug + '-nextjs-15-project.zip';
          a.click();
        }
      }
